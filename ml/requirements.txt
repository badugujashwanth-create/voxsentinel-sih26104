# VoxSentinel ML subsystem (spoof detection) + evaluation.
#
# Kept separate from backend/requirements.txt on purpose: the demo backend must
# stay free of a torch runtime until MLRiskProvider lands.
#
# Install on a clean machine:
#
#   python -m venv ml/.venv
#   source ml/.venv/bin/activate            # Windows: ml\.venv\Scripts\activate
#   python -m pip install --upgrade pip
#   python -m pip install -r ml/requirements.txt
#
# NOTE ON torch: the line below resolves to the CUDA build on PyPI, which pulls
# several GB of NVIDIA wheels this project never uses. Everything here runs on
# CPU. To get the small CPU build instead, install torch first:
#
#   python -m pip install --index-url https://download.pytorch.org/whl/cpu torch==2.9.1
#   python -m pip install -r ml/requirements.txt
#
# pip will then see torch as already satisfied. Both paths work; the CPU one is
# roughly 800 MB instead of 3 GB.

# --- Detector runtime -------------------------------------------------------
torch==2.9.1
numpy==2.3.4
scipy==1.16.3
soundfile==0.13.1

# --- Speaker-verification runtime (ml/speaker/) -----------------------------
# SpeechBrain supplies the ECAPA-TDNN definition and the feature front end that
# ml/scripts/setup_ecapa.py's checkpoint loads into. Apache-2.0, same as the
# checkpoint. Pinned: the model is constructed from the released hyperparams.yaml,
# so a toolkit change is a change to the model this project runs.
#
# 1.1.0 or newer is required, not merely preferred: speechbrain <= 1.0.3 calls
# torchaudio.list_audio_backends(), which torchaudio 2.9 removed, so the pair
# below is the compatible one.
speechbrain==1.1.0
# speechbrain imports torchaudio at package import time. Nothing here decodes
# audio with it -- ml/audio/preprocessing.py uses soundfile -- but the import
# must resolve, and its version must match the torch pin above.
torchaudio==2.9.1

# --- Evaluation-set regeneration --------------------------------------------
# Needed by ml/scripts/prepare_eval_set.py to re-synthesise the 40 synthetic
# samples from the committed manifest. The detector itself never imports piper:
# nothing under ml/spoof/ or ml/audio/ depends on it.
#
# Licence note: piper-tts is GPL-3.0. It is used as a data-generation tool to
# rebuild evaluation audio, not linked into anything VoxSentinel ships. Drop
# this line if you only need to run the detector and the committed metrics.
piper-tts==1.8.0
# Pinned explicitly: piper declares onnxruntime<2,>=1, which leaves the inference
# runtime floating. ONNX Runtime's execution plan affects float results, so an
# unpinned version is an uncontrolled variable in any reproducibility claim.
onnxruntime==1.30.0

# --- Local inference service ------------------------------------------------
fastapi==0.141.1
uvicorn[standard]==0.53.0

# --- Tests ------------------------------------------------------------------
pytest==9.1.1
